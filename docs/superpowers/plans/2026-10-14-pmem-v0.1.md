### 任务 0.1：录制 Hook 脚本

**文件**：创建 `spike/cursor_hooks/record.py`

```python
"""录制 Cursor Hook 的原始输入，仅用于 M0 实测。

用法：python spike/cursor_hooks/record.py <事件名>
- 保存 {事件名, 工作目录, 解释器, 原始输入} 到 spike/cursor_hooks/captures/<毫秒时间戳>_<事件名>.json
- 对 preToolUse：若目标是 pmem-spike 的 echo 工具，返回 updated_input 注入 injected 字段
- 其余事件一律输出 {}，不干预
"""
from __future__ import annotations

import json
import os
import sys
import time
from pathlib import Path

CAPTURE_DIR = Path(__file__).resolve().parent / "captures"


def _find_tool_input(data: dict) -> tuple[str, dict] | None:
    for key in ("tool_input", "arguments", "input", "params"):
        value = data.get(key)
        if isinstance(value, dict):
            return key, value
    return None


def _is_echo_tool(data: dict) -> bool:
    blob = json.dumps({k: data.get(k) for k in ("tool_name", "toolName", "name", "tool")}, ensure_ascii=False)
    return "echo" in blob.lower()


def main() -> None:
    event = sys.argv[1] if len(sys.argv) > 1 else "unknown"
    raw = sys.stdin.read()
    CAPTURE_DIR.mkdir(parents=True, exist_ok=True)
    stamp = int(time.time() * 1000)
    record = {"event": event, "cwd": os.getcwd(), "python": sys.executable, "raw": raw}
    (CAPTURE_DIR / f"{stamp}_{event}.json").write_text(
        json.dumps(record, ensure_ascii=False, indent=2), encoding="utf-8"
    )

    if event == "preToolUse":
        try:
            data = json.loads(raw or "{}")
        except json.JSONDecodeError:
            data = {}
        found = _find_tool_input(data)
        if found and _is_echo_tool(data):
            _, tool_input = found
            updated = dict(tool_input)
            updated["injected"] = "FROM_HOOK"
            print(json.dumps({"permission": "allow", "updated_input": updated}, ensure_ascii=False))
            return
    print("{}")


if __name__ == "__main__":
    main()
```

```python
from __future__ import annotations

import os
import threading
import time

ALPHABET = "0123456789ABCDEFGHJKMNPQRSTVWXYZ"
_RANDOM_BITS = 80
_lock = threading.Lock()
_last_ms = -1
_last_rand = 0


def _encode(value: int, length: int) -> str:
    chars = []
    for _ in range(length):
        value, rem = divmod(value, 32)
        chars.append(ALPHABET[rem])
    return "".join(reversed(chars))


def new_ulid(now_ms: int | None = None) -> str:
    """生成单调递增的 ULID：同一毫秒内随机部分加一，时钟回拨时沿用上一毫秒。"""
    global _last_ms, _last_rand
    ms = int(time.time() * 1000) if now_ms is None else now_ms
    with _lock:
        if ms <= _last_ms:
            ms = _last_ms
            rand = (_last_rand + 1) & ((1 << _RANDOM_BITS) - 1)
        else:
            rand = int.from_bytes(os.urandom(_RANDOM_BITS // 8), "big")
        _last_ms, _last_rand = ms, rand
    return _encode(ms, 10) + _encode(rand, 16)
```

```python
from __future__ import annotations

import re
from dataclasses import dataclass, field

import yaml

KINDS = ("conclusion", "exclusion", "correction", "risk", "decision")
STATUSES = ("draft", "seed", "active", "archived")

_FRONT_MATTER = re.compile(r"\A---\r?\n(.*?)\r?\n---\r?\n?(.*)\Z", re.S)


@dataclass
class Entry:
    id: str
    module: str
    entity: str
    kind: str
    content: str
    note_date: str
    author: str
    revision: str = "none"
    status: str = "active"
    supersedes: list[str] = field(default_factory=list)
    anchors: list[str] = field(default_factory=list)
    source_conversation: str | None = None

    def __post_init__(self) -> None:
        if self.kind not in KINDS:
            raise ValueError(f"kind 必须是 {' / '.join(KINDS)} 之一，当前为 {self.kind!r}")
        if self.status not in STATUSES:
            raise ValueError(f"status 必须是 {' / '.join(STATUSES)} 之一，当前为 {self.status!r}")

    def to_markdown(self) -> str:
        meta: dict = {
            "id": self.id,
            "module": self.module,
            "entity": self.entity,
            "kind": self.kind,
            "status": self.status,
            "supersedes": list(self.supersedes),
            "anchors": list(self.anchors),
            "revision": self.revision,
            "author": self.author,
            "note_date": self.note_date,
        }
        if self.source_conversation:
            meta["source"] = {"conversation": self.source_conversation}
        head = yaml.safe_dump(meta, allow_unicode=True, sort_keys=False).rstrip()
        return f"---\n{head}\n---\n{self.content.strip()}\n"

    @classmethod
    def from_markdown(cls, text: str) -> Entry:
        match = _FRONT_MATTER.match(text)
        if not match:
            raise ValueError("记忆文件缺少 YAML 头（以 --- 开始和结束）")
        meta = yaml.safe_load(match.group(1)) or {}
        source = meta.get("source") or {}
        return cls(
            id=str(meta["id"]),
            module=str(meta["module"]),
            entity=str(meta["entity"]),
            kind=str(meta["kind"]),
            content=match.group(2).replace("\r\n", "\n").strip(),
            note_date=str(meta["note_date"]),
            author=str(meta.get("author") or ""),
            revision=str(meta.get("revision") or "none"),
            status=str(meta.get("status") or "active"),
            supersedes=[str(x) for x in meta.get("supersedes") or []],
            anchors=[str(x) for x in meta.get("anchors") or []],
            source_conversation=source.get("conversation"),
        )
```

**实现**：创建 `src/pmem/core/registry.py`

```python
from __future__ import annotations

import re
from dataclasses import dataclass, field
from pathlib import Path

import yaml

_NAME = re.compile(r"^[A-Za-z][A-Za-z0-9_]*$")
_MERGED_PREFIX = "merged_into:"


class RegistryError(ValueError):
    pass


@dataclass
class Module:
    id: str
    category: str
    name: str
    paths: list[str] = field(default_factory=list)
    aliases: list[str] = field(default_factory=list)
    entities: list[str] = field(default_factory=list)
    components: list[str] = field(default_factory=list)
    variants: list[str] = field(default_factory=list)
    owners: list[str] = field(default_factory=list)
    parent: str | None = None
    origin: str = "manual"
    status: str = "active"

    @property
    def merged_into(self) -> str | None:
        return self.status[len(_MERGED_PREFIX):] if self.status.startswith(_MERGED_PREFIX) else None


@dataclass(frozen=True)
class ResolvedModule:
    id: str
    category: str
    name: str
    known: bool
    candidate: bool = False
    redirected_from: str | None = None


def _split(module_id: str) -> tuple[str, str]:
    category, _, name = module_id.partition(".")
    return category, name


def _build_module(module_id: str, meta: dict | None, *, candidate: bool = False) -> Module:
    meta = meta or {}
    category, name = _split(module_id)
    return Module(
        id=module_id,
        category=category,
        name=name,
        paths=list(meta.get("paths") or []),
        aliases=list(meta.get("aliases") or []),
        entities=list(meta.get("entities") or []),
        components=list(meta.get("components") or []),
        variants=list(meta.get("variants") or []),
        owners=list(meta.get("owners") or []),
        parent=meta.get("parent"),
        origin=str(meta.get("origin") or ("scan" if candidate else "manual")),
        status=str(meta.get("status") or "active"),
    )


class ModuleRegistry:
    def __init__(self, data: dict) -> None:
        self.categories: dict[str, dict] = dict(data.get("categories") or {})
        self.modules: dict[str, Module] = {
            mid: _build_module(mid, meta) for mid, meta in (data.get("modules") or {}).items()
        }
        self.candidates: dict[str, Module] = {
            mid: _build_module(mid, meta, candidate=True) for mid, meta in (data.get("candidates") or {}).items()
        }
        self.third_party: list[str] = list(data.get("third_party") or [])
        self.ignored: list[str] = [
            item["path"] if isinstance(item, dict) else str(item) for item in data.get("ignored") or []
        ]
        self._entity_vocab: dict[str, str] = {}
        for canonical, aliases in (data.get("entities") or {}).items():
            for word in [canonical, *(aliases or [])]:
                self._entity_vocab[str(word).strip().lower()] = canonical
        self._full_index: dict[str, str] = {}
        self._bare_index: dict[str, set[str]] = {}
        for module in self.modules.values():
            self._index(module.id, module.id)
            for alias in module.aliases:
                self._index(alias, module.id)

    @classmethod
    def load(cls, path: Path) -> ModuleRegistry:
        with path.open(encoding="utf-8") as f:
            return cls(yaml.safe_load(f) or {})

    def _index(self, name: str, target: str) -> None:
        key = name.strip().lower()
        if "." in key:
            self._full_index[key] = target
            key = key.split(".", 1)[1]
        self._bare_index.setdefault(key, set()).add(target)

    def _allowed_categories(self) -> str:
        return " / ".join(self.categories)

    def resolve_module(self, raw: str) -> ResolvedModule:
        name = (raw or "").strip()
        if not name:
            raise RegistryError("module 不能为空，格式为 category.Name")
        target = self._lookup(name)
        if target is not None:
            return self._follow_merge(target, original=name)
        if "." not in name:
            raise RegistryError(f"找不到模块 {name!r}；module 格式为 category.Name，例如 gameplay.Jigsaw")
        category, mod_name = name.split(".", 1)
        canonical_category = next((c for c in self.categories if c.lower() == category.lower()), None)
        if canonical_category is None:
            raise RegistryError(f"未知大类 {category!r}，只允许：{self._allowed_categories()}")
        if not _NAME.match(mod_name):
            raise RegistryError(f"模块名 {mod_name!r} 不合法：须为英文字母开头，只含字母、数字、下划线")
        full = f"{canonical_category}.{mod_name}"
        candidate = next((c for c in self.candidates.values() if c.id.lower() == full.lower()), None)
        if candidate is not None:
            return ResolvedModule(candidate.id, candidate.category, candidate.name, known=False, candidate=True)
        return ResolvedModule(full, canonical_category, mod_name, known=False)

    def _lookup(self, name: str) -> str | None:
        key = name.lower()
        if "." in key:
            return self._full_index.get(key)
        targets = self._bare_index.get(key) or set()
        if len(targets) > 1:
            raise RegistryError(f"名字 {name!r} 对应多个模块：{', '.join(sorted(targets))}，请写完整的 category.Name")
        return next(iter(targets), None)

    def _follow_merge(self, target: str, original: str) -> ResolvedModule:
        seen: set[str] = set()
        current = self.modules[target]
        while current.merged_into:
            if current.id in seen or current.merged_into not in self.modules:
                raise RegistryError(f"模块 {current.id} 的合并目标无效：{current.merged_into}")
            seen.add(current.id)
            current = self.modules[current.merged_into]
        redirected = original if current.id.lower() != original.lower() and seen else None
        return ResolvedModule(current.id, current.category, current.name, known=True, redirected_from=redirected)

    def resolve_entity(self, module_id: str, raw: str) -> str:
        entity = (raw or "").strip()
        key = entity.lower()
        module = self.modules.get(module_id) or self.candidates.get(module_id)
        if module is not None:
            for word in [*module.entities, *module.components, *module.variants]:
                if word.lower() == key:
                    return word
        return self._entity_vocab.get(key, entity)
```

```python
_GLOB_FLAGS = wglob.GLOBSTAR | wglob.BRACE
_WILDCARD = re.compile(r"\*\*|\*|\?|\[[^\]]*\]|\{[^}]*\}")


def glob_specificity(pattern: str) -> tuple[int, int]:
    """字面字符越多越具体；字面相同时通配越少越具体。"""
    literal = len(_WILDCARD.sub("", pattern))
    wildcards = len(_WILDCARD.findall(pattern))
    return literal, -wildcards


def _normalise_path(path: str) -> str:
    normalised = path.replace("\\", "/")
    while normalised.startswith("./"):
        normalised = normalised[2:]
    return normalised.lstrip("/")


def _matches(path: str, pattern: str) -> bool:
    pattern = _normalise_path(pattern)
    if not _WILDCARD.search(pattern):
        return path == pattern or path.startswith(pattern.rstrip("/") + "/")
    return wglob.globmatch(path, pattern, flags=_GLOB_FLAGS)


@dataclass(frozen=True)
class PathMatch:
    module_id: str
    pattern: str
    candidate: bool = False
```

在 `ModuleRegistry` 类末尾增加：

```python
    def module_for_path(self, path: str) -> PathMatch | None:
        rel = _normalise_path(path)
        if any(_matches(rel, p) for p in [*self.third_party, *self.ignored]):
            return None
        best: list[tuple[tuple[int, int], PathMatch]] = []
        pools = [(self.modules.values(), False), (self.candidates.values(), True)]
        for modules, is_candidate in pools:
            for module in modules:
                if module.merged_into:
                    continue
                for pattern in module.paths:
                    if _matches(rel, pattern):
                        best.append((glob_specificity(pattern), PathMatch(module.id, pattern, is_candidate)))
        if not best:
            return None
        best.sort(key=lambda item: item[0], reverse=True)
        top_score = best[0][0]
        winners = {m.module_id for score, m in best if score == top_score}
        if len(winners) > 1:
            raise RegistryError(f"路径 {rel} 同时映射到 {', '.join(sorted(winners))}，请在 modules.yaml 中消除重叠")
        return best[0][1]
```
